9.6 Escriba un programa de JavaScript completo para pedir al usuario el radio de una esfera y luego llame a la función volumenEsfera para calcular y mostrar el volumen de la esfera:

Solución completa en HTML5 con JavaScript incorporado: 
<!DOCTYPE html>
<html>
    <head>
        <meta charset = "utf-8">
        <title>Calcular volumen de esferas</title>
        <script>
            function iniciar()
            {
                var boton = document.getElementById( "botonCalcular" );
                boton.addEventListener( "click", mostrarVolumen, false );
            }

            function mostrarVolumen()
            {
                var campoEntrada = document.getElementById( "camporadio" );
                var radio = parseFloat( campoEntrada.value );
                var resultado = document.getElementById( "resultado" );
                resultado.innerHTML = "El volumen de la esfera es: " + volumenEsfera( radio );
            }

            function volumenEsfera( radio )
            {
                return ( 4.0 / 3.0 ) * Math.PI * Math.pow( radio, 3 );
            }

            window.addEventListener( "load", iniciar, false );
        </script>
    </head>
    <body>
        <form action = "#">
            <p><label>Radio:
                <input id = "camporadio" type = "number"></label>
                <input id = "botonCalcular" type = "button" value = "Calcular"></p>
        </form>
        <p id = "resultado"></p>
    </body>
</html>
